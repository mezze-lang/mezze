# Single source of truth for native-image flags shared between:
#   - the `mez` CLI's own native build (cli/build.gradle.kts, graalvmNative.binaries)
#   - the user-app native build path  (cli/.../NativeBuildPath.kt)
#
# Format:
#   - blank lines and lines starting with '#' are ignored
#   - a line beginning with '[section]' starts a new section
#   - every other line is one native-image argument, verbatim
#
# Both consumers parse this file identically (see NativeImageFlags.kt and
# the `readNativeImageFlags()` helper in cli/build.gradle.kts). Keep new
# flags here, not in the consumer code.

[common]
# Common across both CLI and user-app builds.

-H:+UnlockExperimentalVMOptions
--no-fallback
-H:+ReportExceptionStackTraces
-H:+AddAllCharsets
-H:+TruffleHostInlining

# #421: The datastructures module uses jdk.incubator.vector (SIMD via
# LongVector/DoubleVector — see SeqSumSimd, SeqReduce*SIMDNode,
# VectorOps). Without this add-modules flag, native-image's static
# analysis can't see the incubator module — classes that reference
# LongVector/DoubleVector are silently DCE'd from the image, and the
# corresponding static methods vanish. Symptom on user apps that call
# seq::sum: `NoSuchMethodError: SeqSumSimd.sumLong(LongSeqStorage)`
# at first invocation of the native binary. The JVM launcher already
# passes this flag (see BuildCommand.kt DEFAULT_OPTS); mirroring it
# here keeps JVM and native paths in sync.
--add-modules=jdk.incubator.vector

# Truffle-on-Native-Image: links truffle-runtime-svm.jar into the
# native-image builder, enabling Truffle partial evaluation (JIT) in
# the resulting binary. #401 turned this on for user-app builds after
# the 2026-06 doc's "SubstrateVM cannot deopt from continuations"
# framing was found to be stale — the actual blocker was BUILD-time
# blocklist-method reachability (Direct*Node.execute fallbacks reaching
# HashMap/Object.equals), not a VM-level restriction.
--macro:truffle-svm

# Truffle / polyglot language discovery resources.
-H:IncludeResources=META-INF/truffle-languages/.*
-H:IncludeResources=META-INF/services/.*

# Stdlib fallback sources — used only if the CBOR snapshot is missing
# (older builds, debug). Production reads the snapshot.
-H:IncludeResources=stdlib/.*\.mz
-H:IncludeResources=stdlib/MANIFEST.txt

# SQLite-free stdlib snapshot. PrecompiledStdlib.tryLoadFromSnapshot
# reads this on boot — replaces sqlite-jdbc in the runtime.
-H:IncludeResources=META-INF/mezze/stdlib-snapshot\.cbor

# This file itself must ship inside the `mez` native binary — the
# user-app build path (NativeBuildPath) loads it off the classpath
# at user-build time to assemble the native-image command. Without
# this include, `mez build --native` errors with "broken `mez` install".
-H:IncludeResources=META-INF/mezze/native-image-flags\.txt

# Defer init of Netty's BouncyCastle ALPN bridge. Netty's static
# initializer does Class.forName("org.bouncycastle.jsse.BCSSLEngine")
# at class-init time to detect whether BC is on the classpath. We
# don't ship BouncyCastle (no use of Netty's TLS via BC), so the
# probe throws ClassNotFoundException during native-image build —
# 50+ lines of stack trace polluting the log. Defer the clinit to
# runtime where the class is genuinely absent and the probe is a
# harmless no-op (Netty falls back to JDK TLS).
--initialize-at-run-time=io.netty.handler.ssl.BouncyCastleAlpnSslUtils

# #421: jdk.incubator.vector — SeqSumSimd, SeqReduce*SIMDNode et al. hold
# static VectorSpecies instances (LongVector.SPECIES_PREFERRED etc.).
# Because those enclosing classes are init-at-build-time (see the
# com.mezze.datastructures entry above), the VectorSpecies objects land
# in the image heap; SVM then rejects them unless the incubator classes
# are also init-at-build-time. Without this, native-image fails with
# "An object of type 'jdk.incubator.vector.LongVector$LongSpecies' was
# found in the image heap. This type, however, is marked for
# initialization at image run time..."
jdk.incubator.vector

# Stdlib snapshot — runs StdlibSnapshotData.<init> in the build JVM, the
# decoded ModuleImportContext / FunctionIR graph lands in image heap.
com.mezze.language.typechecker.StdlibSnapshotData
com.mezze.language.typechecker.module

# Kotlin runtime + serialization (needed for the snapshot decode).
kotlin
kotlinx

# SLF4J + Logback get pulled in transitively; freeze Logger instances
# so they don't trip the heap scanner.
org.slf4j
ch.qos.logback

[cli-only-flags]
# Raw native-image flags that apply ONLY to the CLI binary's own build
# (not the user-app build path NativeBuildPath uses).

# Let the CLI own SIGINT/SIGTERM. By default native-image installs its
# own signal handlers that terminate the process immediately — that
# overrode SigintForwarder (WireRunner.kt), so Ctrl-C in `mez dev` killed
# the parent `mez` before it could forward an INTERRUPT frame to the child
# RuntimeServer, leaving the app running. Disabling native-image's
# handlers (hosted-option `-H:-InstallExitHandlers`; the `=false` CLI form
# is unsupported) lets our sun.misc.Signal handler run: one Ctrl-C
# interrupts the running program gracefully, a second force-kills.
# (CLI-only: user-app binaries have no such forwarder and want defaults.)
-H:-InstallExitHandlers
